# Gramática didática do subconjunto Java usado na Operação Sentinela - Resilient Community Infrastructure.
# É um registro formal; o programa Python não precisa interpretar este arquivo.

programa -> "public" "class" ID "{" metodo_main "}"
metodo_main -> "public" "static" "void" "main" "(" "String" "[" "]" ID ")" bloco
bloco -> "{" lista_stmt "}"
lista_stmt -> stmt lista_stmt | epsilon
stmt -> declaracao ";" | atribuicao ";" | if_stmt | while_stmt | for_stmt | bloco
declaracao -> "int" ID inicializacao
inicializacao -> "=" expr | epsilon
atribuicao -> ID "=" expr
if_stmt -> "if" "(" condicao ")" stmt
while_stmt -> "while" "(" condicao ")" stmt
for_stmt -> "for" "(" atribuicao ";" condicao ";" atribuicao ")" stmt
condicao -> expr OP_REL expr
OP_REL -> "<" | ">" | "<=" | ">=" | "==" | "!="
expr -> termo expr_linha
expr_linha -> "+" termo expr_linha | "-" termo expr_linha | epsilon
termo -> fator termo_linha
termo_linha -> "*" fator termo_linha | "/" fator termo_linha | epsilon
fator -> NUM | ID | "(" expr ")"
